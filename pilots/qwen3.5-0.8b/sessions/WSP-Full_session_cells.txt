import contextlib
class RowAblate:
    """Row i of the batch has the component along R[i] removed from the residual stream (after the embedding and after every block)."""
    def __init__(self, R): self.R, self.lo, self.nxt = R, 0, 0
    def _rm(self, x):
        r = self.R[self.lo:self.lo + x.shape[0]].to(x.device)
        xf = x.float()
        return (xf - (xf * r[:, None, :]).sum(-1, keepdim=True) * r[:, None, :]).to(x.dtype)
    def embed(self, m, i, out):
        self.lo, self.nxt = self.nxt, self.nxt + out.shape[0]
        return self._rm(out)
    def layer(self, m, i, out):
        return (self._rm(out[0]),) + tuple(out[1:]) if isinstance(out, tuple) else self._rm(out)

@contextlib.contextmanager
def row_ablation(R):
    st, hs = RowAblate(R), []
    hs.append(model.model.embed_tokens.register_forward_hook(st.embed))
    for lyr in model.model.layers: hs.append(lyr.register_forward_hook(st.layer))
    try: yield
    finally:
        for h in hs: h.remove()
